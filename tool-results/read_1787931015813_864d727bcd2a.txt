    67→function Stars({ value, count }: { value: number | null | undefined; count?: number }) {
    68→  if (value === null || value === undefined) {
    69→    return <span className="text-xs text-muted-foreground">No reviews yet</span>;
    70→  }
    71→  return (
    72→    <span className="flex items-center gap-1" aria-label={`Rated ${value} out of 5`}>
    73→      <Star className="size-3.5 fill-amber-400 text-amber-400" />
    74→      <span className="text-xs font-semibold">{value.toFixed(1)}</span>
    75→      {count !== undefined ? <span className="text-xs text-muted-foreground">({count})</span> : null}

   346→// ---------- Payment dialog (post-booking) ----------
   347→
   348→function PaymentDialog({
   349→  appointment,
   350→  open,
   351→  onOpenChange,
   352→  onDone,
   353→}: {
   354→  appointment: AppointmentDTO | null;
   355→  open: boolean;
   356→  onOpenChange: (v: boolean) => void;
   357→  onDone: () => void;
   358→}) {
   359→  const [method, setMethod] = useState("CASH");
   360→  const [paying, setPaying] = useState(false);
   361→
   362→  useEffect(() => {
   363→    if (open) setMethod("CASH");
   364→  }, [open]);
   365→
   366→  async function handlePay() {
   367→    if (!appointment) return;
   368→    setPaying(true);
   369→    try {
   370→      await apiFetch<{ payment: PaymentDTO; appointment: AppointmentDTO }>("/api/payments", {
   371→        method: "POST",
   372→        body: { appointmentId: appointment.id, method },
   373→      });
   374→      toast.success("Payment successful 🎉");
   375→      onOpenChange(false);
   376→      onDone();
   377→    } catch (err) {
   378→      toast.error((err as Error).message);
   379→    } finally {
   380→      setPaying(false);
   381→    }
   382→  }
   383→
   384→  return (
   385→    <Dialog open={open} onOpenChange={onOpenChange}>
   386→      <DialogContent className="sm:max-w-md">
   387→        <DialogHeader>
   388→          <DialogTitle>Pay for your appointment</DialogTitle>
   389→          <DialogDescription>
   390→            {appointment ? `${appointment.service.name} — ${formatBDT(appointment.price)}` : ""}
   391→          </DialogDescription>
   392→        </DialogHeader>
   393→        <RadioGroup value={method} onValueChange={setMethod} className="gap-3">
   394→          {PAYMENT_METHODS.map((pm) => (
   395→            <Label
   396→              key={pm.value}
   397→              htmlFor={`pay-${pm.value}`}
   398→              className={cn(
   399→                "flex min-h-11 cursor-pointer items-center gap-3 rounded-xl border p-3 font-normal transition-colors",
   400→                method === pm.value ? "border-primary bg-primary/5" : "hover:bg-muted/50"
   401→              )}
   402→            >
   403→              <RadioGroupItem value={pm.value} id={`pay-${pm.value}`} />
   404→              {pm.label}
   405→            </Label>
   406→          ))}
   407→        </RadioGroup>
   408→        <DialogFooter>
   409→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={paying} className="min-h-11">
   410→            Cancel
   411→          </Button>
   412→          <Button onClick={handlePay} disabled={paying} className="min-h-11">
   413→            {paying ? <Loader2 className="animate-spin" /> : <BadgeCheck />}
   414→            Pay {appointment ? formatBDT(appointment.price) : ""}
   415→          </Button>
   416→        </DialogFooter>
   417→      </DialogContent>
   418→    </Dialog>
   419→  );
   420→}

   424→export function BookingFlow() {
   425→  const setView = useAppStore((s) => s.setView);
   426→  const [step, setStep] = useState(1);
   427→  const [booked, setBooked] = useState<AppointmentDTO | null>(null);
   428→  const [payOpen, setPayOpen] = useState(false);
   429→
   430→  // selections
   431→  const [serviceId, setServiceId] = useState<string | null>(null);
   432→  const [petId, setPetId] = useState<string | null>(null);
   433→  const [providerId, setProviderId] = useState<string | null>(null);
   434→  const [date, setDate] = useState(todayStr());
   435→  const [time, setTime] = useState<string | null>(null);
   436→  const [notes, setNotes] = useState("");
   437→
   438→  // data
   439→  const [services, setServices] = useState<ServiceDTO[] | null>(null);
   440→  const [pets, setPets] = useState<PetDTO[] | null>(null);
   441→  const [providers, setProviders] = useState<ProviderDTO[] | null>(null);
   442→  const [slots, setSlots] = useState<string[] | null>(null);
   443→  const [slotsLoading, setSlotsLoading] = useState(false);
   444→  const [addPetOpen, setAddPetOpen] = useState(false);
   445→  const [submitting, setSubmitting] = useState(false);
   446→
   447→  const selectedService = services?.find((s) => s.id === serviceId) ?? null;
   448→  const selectedPet = pets?.find((p) => p.id === petId) ?? null;
   449→  const selectedProvider = providers?.find((p) => p.id === providerId) ?? null;
   450→
   451→  useEffect(() => {
   452→    apiFetch<{ services: ServiceDTO[] }>("/api/services?active=true")
   453→      .then((r) => setServices(r.services))
   454→      .catch((err: Error) => toast.error(err.message));
   455→  }, []);
   456→
   457→  useEffect(() => {
   458→    if (step !== 2) return;
   459→    apiFetch<{ pets: PetDTO[] }>("/api/pets")
   460→      .then((r) => setPets(r.pets))
   461→      .catch((err: Error) => toast.error(err.message));
   462→  }, [step, addPetOpen]);
   463→
   464→  useEffect(() => {
   465→    if (step !== 3 || !selectedService) return;
   466→    const specialty = selectedService.category === "GROOMING" ? "GROOMER" : "VET";
   467→    setProviders(null);
   468→    setProviderId(null);
   469→    apiFetch<{ providers: ProviderDTO[] }>(`/api/providers?specialty=${specialty}`)
   470→      .then((r) => setProviders(r.providers))
   471→      .catch((err: Error) => toast.error(err.message));
   472→  }, [step, selectedService]);
   473→
   474→  const fetchSlots = useCallback(async (pid: string, d: string) => {
   475→    setSlotsLoading(true);
   476→    try {
   477→      const r = await apiFetch<{ slots: string[] }>(
   478→        `/api/appointments/slots?providerId=${encodeURIComponent(pid)}&date=${encodeURIComponent(d)}`
   479→      );
   480→      setSlots(r.slots);
   481→    } catch (err) {
   482→      toast.error((err as Error).message);
   483→      setSlots([]);
   484→    } finally {
   485→      setSlotsLoading(false);
   486→    }
   487→  }, []);
   488→
   489→  useEffect(() => {
   490→    if (step !== 4 || !providerId || !date) return;
   491→    setTime(null);
   492→    setSlots(null);
   493→    fetchSlots(providerId, date);
   494→  }, [step, providerId, date, fetchSlots]);
   495→
   496→  const stepValid =
   497→    (step === 1 && !!serviceId) ||
   498→    (step === 2 && !!petId) ||
   499→    (step === 3 && !!providerId) ||
   500→    (step === 4 && !!time) ||
   501→    step === 5;
   502→
   503→  function goNext() {
   504→    if (!stepValid) return;
   505→    setStep((s) => Math.min(s + 1, 5));
   506→  }
   507→
   508→  function goBack() {
   509→    setStep((s) => Math.max(s - 1, 1));
   510→  }
   511→
   512→  async function handleConfirm() {
   513→    if (!serviceId || !petId || !providerId || !time) return;
   514→    setSubmitting(true);
   515→    try {
   516→      const res = await apiFetch<{ appointment: AppointmentDTO }>("/api/appointments", {
   517→        method: "POST",
   518→        body: { petId, serviceId, providerId, date, time, notes: notes.trim() || undefined },
   519→      });
   520→      setBooked(res.appointment);
   521→      toast.success("Appointment booked");
   522→    } catch (err) {
   523→      toast.error((err as Error).message);
   524→    } finally {
   525→      setSubmitting(false);
   526→    }
   527→  }
   528→
   529→  // ---------- success panel ----------
   530→  if (booked) {
   531→    return (
   532→      <motion.div
   533→        initial={{ opacity: 0, scale: 0.97 }}
   534→        animate={{ opacity: 1, scale: 1 }}
   535→        transition={{ duration: 0.35 }}
   536→        className="mx-auto max-w-lg"
   537→      >
   538→        <Card className="gap-0 p-8 text-center">
   539→          <motion.div
   540→            initial={{ scale: 0 }}
   541→            animate={{ scale: 1 }}
   542→            transition={{ delay: 0.15, type: "spring", stiffness: 260, damping: 18 }}
   543→            className="mx-auto flex h-16 w-16 items-center justify-center rounded-full bg-primary/10"
   544→          >
   545→            <Check className="size-9 text-primary" />
   546→          </motion.div>
   547→          <h2 className="mt-4 text-2xl font-bold tracking-tight">Appointment booked!</h2>
   548→          <p className="mt-1 text-sm text-muted-foreground">
   549→            {booked.service.name} for {booked.pet.name} with {booked.provider.name}
   550→          </p>
   551→          <p className="mt-1 flex items-center justify-center gap-1.5 text-sm font-medium">
   552→            <CalendarDays className="size-4 text-primary" />
   553→            {formatDate(booked.date)} at {formatTime(booked.time)}
   554→          </p>
   555→          <div className="mt-2">
   556→            <StatusBadge status={booked.status} />
   557→          </div>
   558→          <div className="mt-6 flex flex-col gap-2 sm:flex-row sm:justify-center">
   559→            <Button onClick={() => setPayOpen(true)} className="min-h-11">
   560→              <BadgeCheck /> Pay now ({formatBDT(booked.price)})
   561→            </Button>
   562→            <Button
   563→              variant="outline"
   564→              onClick={() => {
   565→                toast.info("You can pay anytime from your appointments");
   566→                setView("cust-appointments");
   567→              }}
   568→              className="min-h-11"
   569→            >
   570→              Pay later
   571→            </Button>
